```python
json_output = json.dumps(results, ensure_ascii=False, indent=2 if args.pretty else None)

    )

if args.summary:
    # Compact table to stdout — replaces full JSON when used without --output
    by_country = results['summary'].get('by_country', {})
    header = f"{'Market':<30} {'Errors':>6} {'Warnings':>8} {'Info':>5}"
    print(header)
    print('-' * len(header))
    for country, counts in sorted(by_country.items()):
        print(f"{country:<30} {counts.get('errors', 0):>6} {counts.get('warnings', 0):>8} {counts.get('infos', 0):>5}")
    s = results['summary']
    print('-' * len(header))
    print(f"{'TOTAL':<30} {s['errors']:>6} {s['warnings']:>8} {s['info']:>5}")
elif not args.output:
    # Default: full JSON to stdout (unchanged behavior)
    print(json_output)
```

**Key design decisions:**
- `--summary` without `--output`: prints compact table to stdout (no JSON anywhere)
- `--summary` with `--output`: writes full JSON to file, prints compact table to stdout
- Neither flag with `--output`: writes JSON to file, prints one-line summary to stderr (existing behavior)
- Neither flag without `--output`: prints full JSON to stdout (existing behavior)
- Use `counts.get('infos', 0)` (plural) for by_country dict, but `s['info']` (singular) for the TOTAL row — these keys differ in the existing data structure
